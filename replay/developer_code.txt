  function sprite(id, anim, variant, t, x, y, opt = {}) {
    // (x, y) = 발 아래 가운데

    const sc = opt.scale || 1, w = s.frame_w / PX * sc, h = s.frame_h / PX * sc;
    if (opt.rot || opt.sx !== undefined || opt.sy !== undefined || opt.filter) {
      // 회전·눌림·필터: 그림 가운데를 기준으로 돌리고 줄인다 (발 위치 y는 그대로)
      ctx.save(); ctx.translate(snap(x), snap(y - h / 2)); ctx.rotate(opt.rot || 0);
      ctx.scale((opt.flip ? -1 : 1) * (opt.sx === undefined ? 1 : opt.sx), opt.sy === undefined ? 1 : opt.sy);
      if (opt.filter) ctx.filter = opt.filter;
      ctx.drawImage(im, col * s.frame_w, row * s.frame_h, s.frame_w, s.frame_h, -w / 2, -h / 2, w, h);
      ctx.restore(); return;
    }
    const dx = snap(x - w / 2), dy = snap(y - h);
    if (opt.flip) {
      ctx.save(); ctx.translate(dx + w, dy); ctx.scale(-1, 1);
      ctx.drawImage(im, col * s.frame_w, row * s.frame_h, s.frame_w, s.frame_h, 0, 0, w, h);
      ctx.restore();
    } else {
      ctx.drawImage(im, col * s.frame_w, row * s.frame_h, s.frame_w, s.frame_h, dx, dy, w, h);
    }
  }

    let d = rand() < 0.5 ? -1 : 1;
    for (let i = 0; i < CONFIG.FINAL_FLOOR * CONFIG.STEPS_PER_FLOOR + 40; i++) {
      if (i > 2 && rand() < 0.35) d = -d;
      dirs.push(d);
    }
    G = {
      dirs, step: 0, hearts: CONFIG.HEARTS, days: CONFIG.DAYS_MAX, score: 0, combo: 0, mult: 1, maxMult: 1,
      rankIdx: 0, t: 0, nextAttack: 1.5, attacks: [], facing: 1, pose: "idle", poseT: 0,
      camX: 0, camY: 0, result: null, botCool: 0, flash: 0, popups: [], banner: null,
      crackT: 0, fell: false, nextFlash: CONFIG.FLASH_GAP, camFlash: null,
      mode, parries: 0, hits: 0,
    };
    if (mode === "practice") {                 // 연습: 층 바닥이 있는 칸에 서서 시작, 카메라도 바로 그 자리
      G.step = (CONFIG.PRACTICE_FLOOR - 1) * CONFIG.STEPS_PER_FLOOR;
      const sp = stairPos(G.step); G.camX = sp.x; G.camY = sp.y;
    }
  }

  function moreStairs() {                      // 무한 모드: 계단이 모자라면 더 만든다
    let d = G.dirs[G.dirs.length - 1];
    for (let i = 0; i < 200; i++) { if (rand() < 0.35) d = -d; G.dirs.push(d); }
  }
  const floorOf = (step) => 1 + Math.floor(step / CONFIG.STEPS_PER_FLOOR);
  const zoneOf = (floor) => (floor <= 20 ? 1 : floor <= 40 ? 2 : floor <= 60 ? 3 : floor <= 99 ? 4 : 5);
  const zoneProgress = (floor) => ((floor - 1) % 20) / 19;
  // 계단 i의 월드 좌표 (계단 윗면 가운데)
  function stairPos(i) {
    const st = DEF.sprites && DEF.sprites.stairs;
    const dx = st ? Math.floor(st.frame_w / PX / 2) : 16, dy = st ? st.frame_h / PX : 8;
    let x = 0;
    for (let k = 0; k < i; k++) x += G.dirs[k] * dx;
    return { x, y: -i * dy };
  }

  function hurt(reason) {
    if (G.mode === "practice") {               // 연습: 하트를 잃지 않고 맞은 수만 센다
      G.hits += 1; G.combo = 0; G.mult = 1; G.pose = "hit"; G.poseT = 0.35; G.flash = 0.25;
      play("sfx_hit", 0.6); T.log.push("hurt:" + reason); return;
    }
    G.hearts -= 1; G.combo = 0; G.mult = 1; G.pose = "hit"; G.poseT = 0.35; G.flash = 0.25;
    play("sfx_hit", 0.6); T.log.push("hurt:" + reason);
    if (G.hearts <= 0) endRun("gameover");
  }
  // 캐릭터 위 문구: lane 0 = 패링, lane 1 = 하트 회복 (12px 위). 같은 줄에는 하나만 둔다
  function popup(text, side, lane, color) {
    G.popups = G.popups.filter((q) => q.lane !== lane);

  }
  // 금 간 계단: 21층부터 일부 칸 (층이 바뀌는 칸은 빼고, 칸 번호로 정해져 매번 같다)
  function isCracked(i) {
    if (G && G.mode === "practice") return false;   // 연습 모드에는 계단 장애물이 없다
    const from = G && G.mode === "endless" ? 2 : CONFIG.CRACK_FROM_FLOOR;   // 무한 모드는 2층부터
    if (floorOf(i) < from || i % CONFIG.STEPS_PER_FLOOR === 0) return false;
    return (((i * 2654435761) >>> 0) % 1000) / 1000 < CONFIG.CRACK_SHARE;
  }
  function tryStep(dir) {
    if (screen !== "play" || G.mode === "practice") return;   // 연습: 계단을 오르지 않는다
    if (G.step + 40 > G.dirs.length) moreStairs();
    G.facing = dir;
    if (G.dirs[G.step] === dir) {
      G.step += 1; G.crackT = 0;
      if (isCracked(G.step)) { play("sfx_stair_crack", 0.6); T.log.push("crack:step"); }   // 금 간 계단에 올라섰다
      G.days = Math.min(CONFIG.DAYS_MAX, G.days + CONFIG.STEP_DAYS);
      G.score += CONFIG.STEP_SCORE * G.mult;
      G.pose = "step"; G.poseT = 0.12;
      play("sfx_step", 0.35);
      const f = floorOf(G.step);
      if (f > floorOf(G.step - 1) && G.hearts < CONFIG.HEARTS) { // 새 층 도착: 하트 회복
        G.hearts = Math.min(CONFIG.HEARTS, G.hearts + CONFIG.FLOOR_HEAL);
        popup("하트 +1", 0, 1, PAL.sub);
        T.log.push("heal:floor" + f);
      }
      if (f >= CONFIG.TENSE_FROM_FLOOR) setBgm("bgm_loop_tense");
      if (f > floorOf(G.step - 1) && CONFIG.PROMO_FLOORS.includes(f)) promote(f);
      if (G.mode === "normal" && f >= CONFIG.FINAL_FLOOR && screen === "play") { play("sfx_promote", 0.6); T.log.push("finish:" + f); endRun("promote"); }   // 90층: PoC 마무리
    } else {
      hurt("wrong-step");
    }
  }
  function tryParry(side) { // side: -1 왼쪽(Q), +1 오른쪽(E)
    if (screen !== "play") return;
    G.pose = side < 0 ? "parry_left" : "parry_right"; G.poseT = 0.18;
    // 커피는 막으면 안 된다: 같은 쪽에 막을 공격이 함께 있으면 그것을 먼저 막고, 커피만 있으면 하트 -1
    const inWin = (k) => !k.done && k.side === side && k.locked !== false && G.t >= k.hitAt - CONFIG.PARRY_WINDOW && G.t <= k.hitAt;
    const a = G.attacks.find((k) => inWin(k) && k.kind !== "coffee") || G.attacks.find((k) => inWin(k) && k.kind === "coffee");
    if (!a) return; // 판정 시간 밖의 패링은 효과 없음
    if (a.kind === "coffee") {
      a.done = true; a.parried = true; a.fade = 0.25; T.log.push("coffee:parried");
      popup("커피!", side, 0, PAL.accent); hurt("coffee"); return;
    }
    if (a.kind === "sticky" && a.hp > 1) {     // 포스트잇: 튕겨 나갔다가 같은 쪽으로 돌아온다. 남은 횟수(색)가 하나 준다
      a.hp -= 1; a.spawn = G.t; a.hitAt = G.t + CONFIG.TELEGRAPH; a.bounce = true; a.cued = false; T.log.push("sticky:bounce:" + a.hp);
    } else { a.done = true; a.parried = true; a.fade = 0.25; }
    G.combo += 1; G.parries += 1;
    G.mult = Math.min(CONFIG.MAX_MULT, 1 + Math.floor(G.combo / CONFIG.COMBO_PER_MULT));
    G.maxMult = Math.max(G.maxMult, G.mult);
    G.days = Math.min(CONFIG.DAYS_MAX, G.days + CONFIG.PARRY_DAYS);
    G.score += CONFIG.PARRY_SCORE * G.mult;
    popup(G.mult > 1 ? `×${G.mult}` : "막았다!", side, 0, PAL.warm);
    play("sfx_parry", 0.5);
  }
  function promote(f) {
    G.rankIdx = Math.min(CONFIG.RANKS.length - 1, G.rankIdx + 1);
    G.hearts = Math.min(CONFIG.HEARTS, G.hearts + CONFIG.PROMO_HEAL);
    play("sfx_promote", 0.6);
    T.log.push("promote:" + f + ":" + rankName());
    if (f >= CONFIG.FINAL_FLOOR) { endRun("promote"); return; }
    // 20·40·60·80층: 승급 문구를 띄우고 계속 오른다. 날아오던 장애물은 그대로 둔다 (결정 2026-11-20)
    G.banner = { text: `승급! ${CONFIG.RANKS[G.rankIdx - 1]} → ${rankName()}`, t: 1.6 };
  }
  function endRun(kind) {
    G.result = kind;
    G.pose = kind === "promote" ? "proud" : "box_exit";
    saveBest(floorOf(G.step), G.score);
    if (kind === "gameover") play("sfx_gameover", 0.6);
    go("result");
  }
  // 공격 종류: 결재판(기본), 경쟁 사원(11층~), 전화기(31층~), 서류(41층~)
  function pickKind(f) {
    if (G.mode !== "normal") f = 999;          // 무한·연습: 처음부터 모든 공격
    const w = [["clipboard", 1], ["rival", f >= CONFIG.RIVAL_FROM_FLOOR ? CONFIG.W_RIVAL : 0],
               ["phone", f >= CONFIG.PHONE_FROM_FLOOR ? CONFIG.W_PHONE : 0], ["paper", f >= CONFIG.PAPER_FROM_FLOOR ? CONFIG.W_PAPER : 0],
               ["monitor", f >= CONFIG.MONITOR_FROM_FLOOR ? CONFIG.W_MONITOR : 0],
               ["coffee", f >= CONFIG.COFFEE_FROM_FLOOR ? CONFIG.W_COFFEE : 0], ["sticky", f >= CONFIG.STICKY_FROM_FLOOR ? CONFIG.W_STICKY : 0],
               ["suitcase", f >= CONFIG.SUITCASE_FROM_FLOOR ? CONFIG.W_SUITCASE : 0]];
    let r = rand() * w.reduce((s, x) => s + x[1], 0);
    for (const [k, v] of w) { if (r < v) return k; r -= v; }
    return "clipboard";
  }
  function spawnAttack() {
    const side = rand() < 0.5 ? -1 : 1, f = floorOf(G.step), kind = pickKind(f);
    if (kind === "rival") {
      // 경쟁 사원: 가장 가까운 층 바닥(슬래브)을 걸어 들어와 계단 옆에 멈추고, "!" 1.0초 뒤 플레이어 칸으로 뛰어올라(내려와) 근접 공격 (05-3 요소 5)
      // 플레이어가 그 바닥보다 높이 올라가 있으면(또는 올라가면) 계단 아래 끝에서 계단을 따라 쫓아 올라와, 플레이어 바로 뒤 칸에서 친다.
      // 이때 막는 쪽은 경쟁 사원이 있는 쪽 = 플레이어가 마지막으로 오른 방향의 반대쪽이다 (치기 직전 0.3초에 정해짐)
      G.attacks.push({ side, spawn: G.t, hitAt: G.t + CONFIG.RIVAL_WALK + CONFIG.TELEGRAPH, kind, done: false, locked: false,
                       floorStep: Math.round(G.step / CONFIG.STEPS_PER_FLOOR) * CONFIG.STEPS_PER_FLOOR, // 가장 가까운 층 바닥(위 또는 아래)
                       phase: "walk", wx: G.camX + side * (LW / 2 + 12), ri: 0 });
      T.log.push("rival:spawn");
      return;
    }
    // side는 맞는 쪽(패링 키). 서류는 반대쪽(from)에서 던져져 머리 위로 넘어온다
    // 모니터는 형광등이 먼저 깜박인 뒤(FLICKER) 유리가 깨지며 나타난다. spin·wob·ph는 불규칙한 회전 값
    const lead = kind === "monitor" ? CONFIG.FLICKER : 0;
    G.attacks.push({ side, from: kind === "paper" ? -side : side, warnAt: G.t, spawn: G.t + lead, hitAt: G.t + lead + CONFIG.TELEGRAPH,
                     kind, done: false, spin: (rand() < 0.5 ? -1 : 1) * (4 + rand() * 6), wob: 1.5 + rand() * 3, ph: rand() * 6.28,
                     hp: kind === "sticky" ? 1 + Math.floor(rand() * CONFIG.STICKY_MAX) : 1,
                     debris: rand() < 0.5 ? "document" : "clipboard" });   // 서류가방에서 쏟아질 것
    T.log.push(kind + ":spawn");
    if (kind === "monitor") play("sfx_light_flicker", 0.6);     // 형광등 깜박이는 소리
    else if (kind !== "phone" && kind !== "coffee") play("sfx_warning", 0.3);   // 커피는 경고음 없이 높은 "헤이!"만
  }

  function updatePlay(dt) {
    G.t += dt;
    const floor = floorOf(G.step);
    // 난이도: 보통은 20층 구간마다, 무한은 60층에 걸쳐 끝까지, 연습은 가운데 값으로 고정
    const p = G.mode === "endless" ? Math.min(1, (floor - 1) / CONFIG.ENDLESS_RAMP_FLOORS) : G.mode === "practice" ? CONFIG.PRACTICE_P : zoneProgress(floor);
    if (G.mode !== "practice") G.days -= (dt / CONFIG.SEC_PER_DAY) * (1 + CONFIG.DRAIN_RAMP * p);   // 연습: 시간 제한 없음
    if (isCracked(G.step)) {                     // 금 간 계단에 너무 오래 서 있으면 무너진다
      G.crackT += dt;
      if (G.crackT >= CONFIG.CRACK_TIME) { G.fell = true; T.log.push("crack:fall"); play("sfx_stair_crack", 0.8); endRun("gameover"); return; }
    }
    if (floor >= CONFIG.FLASH_FROM_FLOOR || G.mode !== "normal") {      // 카메라 플래시 (무한·연습은 처음부터): 막을 수 없지만 피해도 없다
      G.nextFlash -= dt;
      if (G.nextFlash <= 0) {
        G.camFlash = { t: CONFIG.FLASH_TIME, x: 20 + rand() * (LW - 40), y: 20 + rand() * 60 };
        G.nextFlash = CONFIG.FLASH_GAP * (0.6 + rand() * 0.8);
        play("sfx_camera_flash", 0.6); T.log.push("flash");
      }
    }
    if (G.camFlash) { G.camFlash.t -= dt; if (G.camFlash.t <= 0) G.camFlash = null; }
    if (G.days <= 0) { G.days = 0; endRun("gameover"); return; }
    G.nextAttack -= dt;
    if (G.nextAttack <= 0) {
      spawnAttack();
      const gap = CONFIG.ATTACK_GAP_START + (CONFIG.ATTACK_GAP_END - CONFIG.ATTACK_GAP_START) * p;
      G.nextAttack = gap * (0.8 + rand() * 0.4);
    }
    if (G.banner) { G.banner.t -= dt; if (G.banner.t <= 0) G.banner = null; }
    for (const a of G.attacks) {
      if (a.kind === "rival") updateRival(a, dt);
      if (a.kind === "monitor" && !a.shattered && G.t >= a.spawn) { a.shattered = true; play("sfx_monitor_shatter", 0.6); }   // 유리 깨짐 (도착 1.0초 전)
      // 패링 타이밍 소리: 소리의 박이 패링하기 가장 좋은 순간(도착 0.1초 전)에 온다. 경쟁 사원은 막을 쪽이 정해진 뒤에만
      const cue = CONFIG.CUES[a.kind];
      if (cue && !a.cued && !a.done && a.locked !== false && G.t >= a.hitAt - cue[1]) { a.cued = true; play(cue[0], 0.7); T.log.push("cue:" + a.kind); }
      if (!a.done && G.t > a.hitAt) {
        a.done = true; a.fade = 0.2;
        if (a.kind === "coffee") { a.passed = true; T.log.push("coffee:dodged"); }   // 커피는 막지 않고 지나가게 두면 된다
        else { hurt("attack"); if (screen !== "play") return; }
      }
      if (a.fade !== undefined) a.fade -= dt;
    }
    G.attacks = G.attacks.filter((a) => !a.done || a.fade > 0);
    if (G.poseT > 0) { G.poseT -= dt; if (G.poseT <= 0) G.pose = "idle"; }
    G.flash = Math.max(0, G.flash - dt);
    G.popups.forEach((q) => (q.t -= dt)); G.popups = G.popups.filter((q) => q.t > 0);
    // 카메라: 플레이어를 화면 (160, 120)에 둔다
    const sp = stairPos(G.step);
    G.camX += (sp.x - G.camX) * Math.min(1, dt * 12);
    G.camY += (sp.y - G.camY) * Math.min(1, dt * 12);
    if (BOT) botThink(dt);
  }
  // 경쟁 사원 위치: 걷기·서 있기는 층 바닥 위(wx), 오르기는 계단 칸(ri, 소수)
  function rivalPos(a) {
    if (a.phase !== "climb") return { x: a.wx, y: stairPos(a.floorStep).y };
    const i = Math.floor(a.ri), f = a.ri - i, p0 = stairPos(i), p1 = stairPos(i + 1);
    return { x: p0.x + (p1.x - p0.x) * f, y: p0.y + (p1.y - p0.y) * f };
  }
  function updateRival(a, dt) {
    if (a.done) return;
    const me = stairPos(G.step), above = G.step > a.floorStep;
    if (a.phase === "walk" || a.phase === "stand") {
      // 플레이어가 바닥보다 위면 계단 아래 끝으로, 아니면 플레이어 옆으로 걷는다
      const tx = above ? stairPos(a.floorStep).x : me.x + a.side * 16, d = tx - a.wx;
      a.wx += Math.sign(d) * Math.min(Math.abs(d), CONFIG.RIVAL_SPEED * dt);
      a.phase = Math.abs(tx - a.wx) < 0.5 ? (above ? "climb" : "stand") : "walk";
      if (a.phase === "climb") a.ri = a.floorStep;
    } else if (a.ri < G.step - 1) {
      a.ri = Math.min(G.step - 1, a.ri + CONFIG.RIVAL_CLIMB * dt);   // 계단을 따라 쫓아 오른다
    }
    const ready = a.phase === "stand" || (a.phase === "climb" && a.ri >= G.step - 1 - 1e-6);
    if (a.locked) return;
    if (a.phase === "climb" && ready) a.side = -G.dirs[G.step - 1];   // 바로 뒤 칸: 플레이어가 온 쪽
    if (G.t >= a.hitAt - CONFIG.RIVAL_LEAP) {
      if (ready) { a.locked = true; T.log.push("rival:lock" + (a.phase === "climb" ? ":stairs" : ":floor")); }
      else a.hitAt = G.t + CONFIG.RIVAL_LEAP;                         // 아직 따라잡지 못했다: 붙을 때까지 치지 않는다
    }
  }
  function botThink(dt) {
    G.botCool -= dt;
    for (const a of G.attacks) {
      if (!a.done && a.kind !== "coffee" && a.locked !== false && a.hitAt - G.t <= 0.08) { pressKey(a.side < 0 ? "KeyQ" : "KeyE"); }
    }

    else if (code === "ArrowRight" || code === "ArrowDown") focus = (focus + 1) % buttons.length;
    else if (code === "Enter" || code === "Space") pressButton(focus);
  }

  let last = performance.now();
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000); last = now;
    clock += dt;
    for (const k in keyGlow) keyGlow[k] = Math.max(0, keyGlow[k] - dt);
    if (screen === "play" && !settingsOpen) updatePlay(dt);

    ctx.setTransform(S, 0, 0, S, 0, 0);
    ctx.imageSmoothingEnabled = false;
    ctx.clearRect(0, 0, LW, LH);
    if (screen === "title") drawTitle();
    else if (screen === "lobby") drawLobby();
    else if (screen === "play") { drawWorld(); drawHud(); }
    else if (screen === "pause") drawPause();
    else if (screen === "result") drawResult();
    if (settingsOpen) (helpOpen ? drawHelp : drawSettings)();

    if (G) Object.assign(T, {
      floor: floorOf(G.step), rank: rankName(), hearts: G.hearts, days: G.days,
      score: G.score, maxMult: G.maxMult, steps: G.step,
    });
    T.screen = settingsOpen ? screen + (helpOpen ? ":help" : ":settings") : screen;
    requestAnimationFrame(frame);
  }

  function start() {
    if (pending > 0) { setTimeout(start, 30); return; }
    newRun();
    go("title");
    T.ready = true;
    requestAnimationFrame(frame);
  }
  start();
